    <script type="module">
        import './js/api.js';
        import { bindShellActions, showToast } from './js/ui.js';

        if (!window.api.token) window.location.href = '/login.html';
        bindShellActions();

        let fieldToDelete = null;

        window.openModal = function(id) {
            document.getElementById(id).style.display = 'block';
        };
        
        window.closeModal = function(id) {
            document.getElementById(id).style.display = 'none';
        };

        async function loadFields() {
            try {
                const fields = await window.api.getFieldDefinitions();
                const tbody = document.querySelector('#fieldsTable tbody');
                tbody.innerHTML = '';
                
                const typeNames = ['Text', 'Number', 'Date', 'Dropdown', 'MultiSelect'];
                const typeColors = ['info', 'warning', 'success', 'primary', 'default'];

                if (fields.length === 0) {
                    tbody.innerHTML = `<tr><td colspan="5" style="text-align: center; padding: var(--spacing-xl); color: var(--text-muted);">No custom fields found.</td></tr>`;
                    return;
                }

                fields.forEach(f => {
                    const tr = document.createElement('tr');
                    const badgeClass = `badge badge-${typeColors[f.fieldType] || 'default'}`;
                    
                    tr.innerHTML = `
                        <td style="font-family: monospace; font-size: 13px;">${escapeHtml(f.key)}</td>
                        <td style="font-weight: 500;">${escapeHtml(f.label)}</td>
                        <td><span class="${badgeClass}">${typeNames[f.fieldType]}</span></td>
                        <td class="text-muted" style="font-size: 13px;">${escapeHtml(f.validationRegex) || '-'}</td>
                        <td>
                            <button type="button" class="btn btn-ghost" style="color: var(--danger); padding: 4px 8px;" onclick="promptDelete(${f.id})">
                                <svg viewBox="0 0 24 24" width="18" height="18" style="fill: currentColor;"><path d="M6 19c0 1.1.9 2 2 2h8c1.1 0 2-.9 2-2V7H6v12zM19 4h-3.5l-1-1h-5l-1 1H5v2h14V4z"/></svg>
                            </button>
                        </td>
                    `;
                    tbody.appendChild(tr);
                });
            } catch (err) {
                console.error(err);
                showToast('Error loading fields', 'error');
            }
        }

        window.promptDelete = function(id) {
            fieldToDelete = id;
            openModal('confirmModal');
        };

        document.getElementById('btnConfirmDelete').addEventListener('click', async () => {
            if (!fieldToDelete) return;
            try {
                await window.api.request(`/DynamicForm/definitions/${fieldToDelete}`, { method: 'DELETE' });
                closeModal('confirmModal');
                showToast('Field deleted successfully');
                await loadFields();
            } catch (e) {
                console.error(e);
                showToast('Failed to delete field', 'error');
            } finally {
                fieldToDelete = null;
            }
        });

        document.getElementById('btnCreateFieldModal').addEventListener('click', () => {
            document.getElementById('createForm').reset();
            openModal('createModal');
        });

        document.getElementById('createForm').addEventListener('submit', async (e) => {
            e.preventDefault();
            const key = document.getElementById('fKey').value.trim();
            const label = document.getElementById('fLabel').value.trim();
            const typeStr = document.getElementById('fType').value;
            
            try {
                await window.api.request('/DynamicForm/definitions', {
                    method: 'POST',
                    body: JSON.stringify({
                        key: key,
                        label: label,
                        fieldType: Number.parseInt(typeStr) || 0,
                        validationRegex: null
                    })
                });
                closeModal('createModal');
                showToast('Field created successfully');
                await loadFields();
            } catch (e) {
                console.error(e);
                showToast('Failed to create field', 'error');
            }
        });

        function escapeHtml(unsafe) {
            return (unsafe || '').toString()
                .replaceAll('&', "&amp;")
                .replaceAll('<', "&lt;")
                .replaceAll('>', "&gt;")
                .replaceAll('"', "&quot;")
                .replaceAll("'", "&#039;");
        }

        // Close modal when clicking outside
        window.onclick = function(event) {
            if (event.target.classList.contains('modal')) {
                event.target.style.display = "none";
            }
        }

        await loadFields();
    </script>
