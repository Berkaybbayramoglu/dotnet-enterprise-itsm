    <script type="module">
        import './js/api.js';
        import { bindShellActions, showToast } from './js/ui.js';
        
        if (!window.api.token) window.location.href = '/login.html';
        bindShellActions();

        async function loadKb(keyword = '') {
            try {
                const articles = await window.api.getKbArticles(keyword);
                const list = document.getElementById('articleList');
                list.innerHTML = '';
                
                if (articles.length === 0) {
                    list.innerHTML = `
                        <div style="grid-column: 1 / -1; text-align: center; padding: 40px; color: var(--text-muted);">
                            <svg viewBox="0 0 24 24" width="48" height="48" style="fill: var(--border); margin-bottom: 16px;"><path d="M12 2C6.48 2 2 6.48 2 12s4.48 10 10 10 10-4.48 10-10S17.52 2 12 2zm1 15h-2v-2h2v2zm0-4h-2V7h2v6z"/></svg>
                            <h3>No articles found</h3>
                            <p>Try adjusting your search terms.</p>
                        </div>
                    `;
                    return;
                }

                articles.forEach(a => {
                    const btn = document.createElement('button');
                    btn.type = 'button';
                    btn.className = 'article-card';
                    btn.addEventListener('click', () => window.openArticle(a.id));
                    
                    let badges = '';
                    if (a.visibility === 1) badges += '<span class="badge badge-warning">Internal</span> ';
                    if (a.status === 0) badges += '<span class="badge badge-default">Draft</span> ';

                    btn.innerHTML = `
                        <div class="article-card-header">
                            <h3>${escapeHtml(a.title)}</h3>
                            <div style="display:flex; gap: 4px;">${badges}</div>
                        </div>
                        <p style="color: var(--text-muted); font-size: 13px; line-height: 1.5; text-align: left; margin-bottom: 16px; display: -webkit-box; -webkit-line-clamp: 2; -webkit-box-orient: vertical; overflow: hidden;">
                            Read more about ${escapeHtml(a.title)}...
                        </p>
                        <div class="meta">
                            <span class="meta-item"><svg viewBox="0 0 24 24" width="14" height="14" style="fill: currentColor;"><path d="M12 4.5C7 4.5 2.73 7.61 1 12c1.73 4.39 6 7.5 11 7.5s9.27-3.11 11-7.5c-1.73-4.39-6-7.5-11-7.5zM12 17c-2.76 0-5-2.24-5-5s2.24-5 5-5 5 2.24 5 5-2.24 5-5 5zm0-8c-1.66 0-3 1.34-3 3s1.34 3 3 3 3-1.34 3-3-1.34-3-3-3z"/></svg> ${a.viewCount}</span>
                            <span class="meta-item"><svg viewBox="0 0 24 24" width="14" height="14" style="fill: currentColor;"><path d="M19 3h-1V1h-2v2H8V1H6v2H5c-1.11 0-1.99.9-1.99 2L3 19c0 1.1.89 2 2 2h14c1.1 0 2-.9 2-2V5c0-1.1-.9-2-2-2zm0 16H5V8h14v11zM7 10h5v5H7z"/></svg> ${new Date(a.createdAt).toLocaleDateString()}</span>
                        </div>
                    `;
                    list.appendChild(btn);
                });
            } catch(e) { console.error(e); showToast('Search failed', 'error'); }
        }

        window.openArticle = async function(id) {
            try {
                const res = await window.api.request(`/kb/articles/${id}`);
                document.getElementById('modalTitle').textContent = res.title;
                document.getElementById('modalBody').innerHTML = `<p>${escapeHtml(res.content).replace(/\n/g, '<br>')}</p>`;
                document.getElementById('articleModal').style.display = 'block';
                document.body.style.overflow = 'hidden'; // Prevent background scrolling
            } catch(e) {
                console.error(e);
                showToast('Error loading article', 'error');
            }
        };

        window.closeModal = function() {
            document.getElementById('articleModal').style.display = 'none';
            document.getElementById('modalTitle').textContent = 'Loading...';
            document.getElementById('modalBody').innerHTML = '';
            document.body.style.overflow = '';
        };

        function escapeHtml(unsafe) {
            return (unsafe || '').toString()
                .replaceAll('&', "&amp;")
                .replaceAll('<', "&lt;")
                .replaceAll('>', "&gt;")
                .replaceAll('"', "&quot;")
                .replaceAll("'", "&#039;");
        }

        document.getElementById('btnSearchKb').addEventListener('click', async () => {
            const kw = document.getElementById('searchInput').value;
            await loadKb(kw);
        });
        
        // Allow pressing enter to search
        document.getElementById('searchInput').addEventListener('keypress', async (e) => {
            if (e.key === 'Enter') {
                await loadKb(e.target.value);
            }
        });

        document.getElementById('btnCloseModal').addEventListener('click', window.closeModal);
        
        // Close modal when clicking outside
        window.onclick = function(event) {
            const modal = document.getElementById('articleModal');
            if (event.target === modal) {
                window.closeModal();
            }
        }

        await loadKb();
    </script>
