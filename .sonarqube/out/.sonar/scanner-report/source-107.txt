    <script src="https://cdnjs.cloudflare.com/ajax/libs/dompurify/3.0.6/purify.min.js" integrity="sha384-cwS6YdhLI7XS60eoDiC+egV0qHp8zI+Cms46R0nbn8JrmoAzV9uFL60etMZhAnSu" crossorigin="anonymous"></script>
    <script src="https://cdn.quilljs.com/1.3.7/quill.min.js" integrity="sha384-QUJ+ckWz1M+a7w0UfG1sEn4pPrbQwSxGm/1TIPyioqXBrwuT9l4f9gdHWLDLbVWI" crossorigin="anonymous"></script>
    <script type="module">
        import './js/api.js?v=3';
        import { injectShell } from './js/layout.js?v=10';
        import { bindShellActions, showToast, showUndoToast, openModal, closeModal } from './js/ui.js?v=3';
        import { t } from './js/i18n.js';
        injectShell();
        window.closeModal = closeModal;
        
        if (!window.api.token) window.location.href = '/login.html';
        bindShellActions();

        const urlParams = new URLSearchParams(window.location.search);
        const articleId = urlParams.get('id');

        function escapeHtml(unsafe) {
            return (unsafe || '').toString()
                .replaceAll('&', "&amp;")
                .replaceAll('<', "&lt;")
                .replaceAll('>', "&gt;")
                .replaceAll('"', "&quot;")
                .replaceAll("'", "&#039;");
        }

        async function loadArticle() {
            if (!articleId) {
                showToast('Article ID missing', 'error');
                return;
            }
            try {
                const a = await window.api.request(`/kb/articles/${articleId}`);
                document.title = `${a.title} - ITSM Knowledge Base`;
                document.getElementById('articleTitle').textContent = a.title;
                document.getElementById('articleDate').textContent = new Date(a.createdAt).toLocaleString('tr-TR', { year: 'numeric', month: '2-digit', day: '2-digit', hour: '2-digit', minute: '2-digit' });
                
                if (a.authorName) {
                    document.getElementById('articleAuthor').style.display = 'inline';
                    document.getElementById('articleAuthor').innerHTML = ` • Gönderen: ${escapeHtml(a.authorName)}${a.authorDepartment ? ` (${escapeHtml(a.authorDepartment)})` : ''}`;
                }

                document.getElementById('articleViews').textContent = a.viewCount;
                
                let badges = '';
                if (a.visibility === 1) badges += `<span class="badge badge-warning" style="margin-right: 8px;">${t('kb_badge_internal') || 'Internal'}</span>`;
                if (a.status === 0) badges += `<span class="badge badge-default">${t('kb_badge_draft') || 'Draft'}</span>`;
                if (a.status === 2) badges += `<span class="badge badge-info" style="background: var(--info); color: white;">${t('kb_badge_pending') || 'Pending Review'}</span>`;
                if (a.status === 3) badges += `<span class="badge badge-warning" style="background: rgba(var(--warning-rgb, 245,158,11), 0.2); color: var(--warning);">${t('kb_badge_needs_rev') || 'Needs Revision'}</span>`;
                if (a.status === 4) badges += `<span class="badge badge-danger">${t('kb_badge_rejected') || 'Rejected'}</span>`;
                document.getElementById('articleBadges').innerHTML = badges;
                
                if ((a.status === 3 || a.status === 4) && a.managerFeedback) {
                    document.getElementById('pageFeedbackAlert').style.display = 'block';
                    document.getElementById('pageFeedbackAlert').style.background = 'rgba(var(--warning-rgb, 245,158,11), 0.1)';
                    document.getElementById('pageFeedbackAlert').style.borderLeftColor = 'var(--warning)';
                    document.getElementById('pageFeedbackTitle').style.color = 'var(--warning)';
                    document.getElementById('pageFeedbackTitle').textContent = 'Manager Feedback / Revision Request';
                    document.getElementById('pageFeedbackText').textContent = a.managerFeedback;
                } else if (a.status === 2 && a.managerFeedback) {
                    document.getElementById('pageFeedbackAlert').style.display = 'block';
                    document.getElementById('pageFeedbackAlert').style.background = 'rgba(var(--info-rgb, 59,130,246), 0.1)';
                    document.getElementById('pageFeedbackAlert').style.borderLeftColor = 'var(--info)';
                    document.getElementById('pageFeedbackTitle').style.color = 'var(--info)';
                    document.getElementById('pageFeedbackTitle').textContent = '📌 Yazar Revizyonu Tamamladı! (Önceki İnceleme Notunuz Aşağıdadır)';
                    document.getElementById('pageFeedbackText').textContent = a.managerFeedback;
                } else {
                    document.getElementById('pageFeedbackAlert').style.display = 'none';
                }

                document.getElementById('articleContent').innerHTML = typeof DOMPurify !== 'undefined' ? DOMPurify.sanitize(a.content) : escapeHtml(a.content);
                loadedArticle = a;
                const isAuthor = currentUserId > 0 && a.authorUserId === currentUserId;

                if (isManager || (isAuthor && a.status !== 1)) {
                    document.getElementById('adminActionBar').style.display = 'flex';
                    await loadCategoriesForEdit();
                } else {
                    document.getElementById('adminActionBar').style.display = 'none';
                }

                if(isManager) {
                    document.getElementById('managerTip').style.display = 'flex';
                    const tBtn = document.getElementById('btnTogglePublish');
                    tBtn.textContent = a.status === 0 ? (t('kb_btn_pub') || 'Publish') : (t('kb_btn_unpub') || 'Unpublish');
                    if (a.status === 2) {
                        tBtn.style.display = 'none';
                        document.getElementById('btnReviewArticle').style.display = 'inline-flex';
                    } else {
                        tBtn.style.display = 'inline-flex';
                        document.getElementById('btnReviewArticle').style.display = 'none';
                    }
                } else {
                    document.getElementById('managerTip').style.display = 'none';
                    document.getElementById('btnTogglePublish').style.display = 'none';
                    document.getElementById('btnReviewArticle').style.display = 'none';
                }
            } catch (e) { console.error(e); 
                showToast(t('kb_toast_load_err') || 'Failed to load article', 'error');
                document.getElementById('articleTitle').textContent = t('kb_not_found') || 'Article Not Found';
                document.getElementById('articleContent').innerHTML = `<p class="text-danger">${t('kb_not_found_desc') || 'The article you are looking for does not exist or you do not have permission to view it.'}</p>`;
             }
        }
        
        
        let loadedArticle = null;
        let isManager = false;
        let currentUserId = 0;

        try {
            const payload = JSON.parse(atob(window.api.token.split('.')[1]));
            let currentUserIdStr = payload['http://schemas.xmlsoap.org/ws/2005/05/identity/claims/nameidentifier'] || payload.nameid || payload.sub || '0';
            currentUserId = Number.parseInt(currentUserIdStr, 10);
            
            let perms = [];
            let permsClaim = payload.permission || payload.Permissions;
            if (permsClaim) {
                perms = typeof permsClaim === 'string' ? [permsClaim] : permsClaim;
            }
            if (perms.includes('kb.manage')) {
                isManager = true;
            }
        } catch (e) { console.error(e); }

        async function loadCategoriesForEdit() {
            if (document.getElementById('editArticleCategory').options.length === 0) {
                try {
                    const cats = await window.api.request('/kb/categories');
                    const sel = document.getElementById('editArticleCategory');
                    cats.forEach(c => {
                        sel.innerHTML += `<option value="${c.id}">${escapeHtml(c.name)}</option>`;
                    });
                } catch (e) { console.error(e); }
            }
        }

        let quill = null;
        function initQuill() {
            if (!quill) {
                quill = new Quill('#quillEditor', {
                    theme: 'snow',
                    modules: {
                        toolbar: [
                            [{ 'header': [1, 2, 3, false] }],
                            ['bold', 'italic', 'underline', 'strike'],
                            [{ 'color': [] }, { 'background': [] }],
                            [{ 'list': 'ordered'}, { 'list': 'bullet' }],
                            ['link', 'image'],
                            ['clean']
                        ]
                    }
                });
                quill.on('text-change', function() {
                    document.getElementById('editArticleContent').value = quill.root.innerHTML === '<p><br></p>' ? '' : quill.root.innerHTML;
                });
            }
        }

        document.getElementById('btnEditArticle')?.addEventListener('click', () => {
            if(!loadedArticle) return;
            document.getElementById('editArticleTitle').value = loadedArticle.title;
            document.getElementById('editArticleCategory').value = loadedArticle.categoryId;
            document.getElementById('editArticleVisibility').value = loadedArticle.visibility;
            
            const statusSel = document.getElementById('editArticleStatus');
            if (!isManager) {
                statusSel.innerHTML = `
                    <option value="0">${t('kb_opt_draft') || 'Draft'}</option>
                    <option value="2">Gönder (Onaya Sun)</option>
                `;
                if (loadedArticle.status === 3) {
                    statusSel.value = "2";
                } else if (loadedArticle.status === 0) {
                    statusSel.value = "0";
                } else {
                    statusSel.value = "2";
                }
            } else {
                statusSel.innerHTML = `
                    <option value="0">${t('kb_opt_draft') || 'Draft'}</option>
                    <option value="1">${t('kb_opt_published') || 'Published'}</option>
                    <option value="2">${t('kb_badge_pending') || 'Pending Review'}</option>
                    <option value="3">${t('kb_badge_needs_rev') || 'Needs Revision'}</option>
                `;
                statusSel.value = loadedArticle.status;
            }

            initQuill();
            quill.root.innerHTML = loadedArticle.content || '';
            document.getElementById('editArticleContent').value = loadedArticle.content || '';

            if (loadedArticle.status === 3 && loadedArticle.managerFeedback) {
                document.getElementById('authorFeedbackAlert').style.display = 'block';
                document.getElementById('authorFeedbackText').textContent = loadedArticle.managerFeedback;
                document.getElementById('btnSaveArticle').textContent = t('kb_btn_fix') || 'Fix & Resubmit';
            } else {
                document.getElementById('authorFeedbackAlert').style.display = 'none';
                document.getElementById('btnSaveArticle').textContent = t('kb_btn_save') || 'Save';
            }

            openModal('articleModal');
        });

        document.getElementById('btnSaveArticle')?.addEventListener('click', async () => {
            const data = {
                title: document.getElementById('editArticleTitle').value,
                categoryId: Number.parseInt(document.getElementById('editArticleCategory').value, 10),
                visibility: Number.parseInt(document.getElementById('editArticleVisibility').value, 10),
                status: Number.parseInt(document.getElementById('editArticleStatus').value, 10),
                content: document.getElementById('editArticleContent').value
            };
            try {
                await window.api.request(`/kb/articles/${articleId}`, { method: 'PUT', body: JSON.stringify(data) });
                closeModal('articleModal');
                showToast('Article updated successfully.');
                loadArticle();
            } catch (e) { console.error(e);  showToast('Update failed', 'error');  }
        });

        document.getElementById('btnTogglePublish')?.addEventListener('click', async () => {
            if(!loadedArticle) return;
            const newStatus = loadedArticle.status === 0 ? 1 : 0;
            const oldStatus = loadedArticle.status;
            
            try {
                loadedArticle.status = newStatus;
                await window.api.request(`/kb/articles/${articleId}`, { method: 'PUT', body: JSON.stringify(loadedArticle) });
                loadArticle(); // Refresh UI
                
                showUndoToast(newStatus === 1 ? (t('kb_toast_pub') || 'Article published.') : (t('kb_toast_unpub') || 'Article unpublished.'), async () => {
                    loadedArticle.status = oldStatus;
                    await window.api.request(`/kb/articles/${articleId}`, { method: 'PUT', body: JSON.stringify(loadedArticle) });
                    loadArticle();
                });
            } catch (e) { console.error(e); 
                loadedArticle.status = oldStatus;
                showToast('Failed to change publish status.', 'error');
             }
        });

        document.getElementById('btnReviewArticle')?.addEventListener('click', () => {
            if(!loadedArticle) return;
            document.getElementById('reviewStatus').value = '1';
            document.getElementById('reviewFeedback').value = '';
            document.getElementById('feedbackGroup').style.display = 'none';
            openModal('reviewModal');
        });

        document.getElementById('reviewStatus')?.addEventListener('change', (e) => {
            if (e.target.value === '3' || e.target.value === '4') {
                document.getElementById('feedbackGroup').style.display = 'block';
            } else {
                document.getElementById('feedbackGroup').style.display = 'none';
            }
        });

        document.getElementById('btnSubmitReview')?.addEventListener('click', async () => {
            if(!loadedArticle) return;
            const form = document.getElementById('reviewForm');
            if (!form.checkValidity()) { form.reportValidity(); return; }

            const payload = {
                status: Number.parseInt(document.getElementById('reviewStatus').value, 10),
                feedback: document.getElementById('reviewFeedback').value
            };

            try {
                await window.api.reviewKbArticle(articleId, payload);
                showToast(t('kb_toast_updated') || 'Article reviewed successfully.');
                closeModal('reviewModal');
                loadArticle();
            } catch (e) {
                console.error(e);
                showToast(t('kb_toast_save_err') || 'Failed to submit review', 'error');
            }
        });

        document.getElementById('btnDeleteArticle')?.addEventListener('click', () => {
            showUndoToast('Article will be deleted.', async () => {
                // Cancelled
                console.log('Delete cancelled by Undo.');
            }).then(async (proceed) => {
                if (proceed) {
                    try {
                        await window.api.request(`/kb/articles/${articleId}`, { method: 'DELETE' });
                        window.location.href = '/kb.html';
                    } catch (e) { console.error(e);  showToast('Delete failed.', 'error');  }
                }
            });
        });


        // --- Inline Text Selection Logic (Overleaf-style) ---
        let selectedTextForReview = '';

        document.getElementById('articleContent').addEventListener('mouseup', (e) => {
            if (!isManager) return;
            const selection = window.getSelection();
            const text = selection.toString().trim();
            
            if (text.length > 0) {
                selectedTextForReview = text;
                const range = selection.getRangeAt(0);
                const rect = range.getBoundingClientRect();
                const floatingBtn = document.getElementById('floatingReviewBtn');
                
                // Position above the selection (adjusted for scrolling)
                floatingBtn.style.top = (window.scrollY + rect.top - 45) + 'px';
                floatingBtn.style.left = (window.scrollX + rect.left + (rect.width / 2) - 80) + 'px';
                floatingBtn.style.display = 'block';
            } else {
                document.getElementById('floatingReviewBtn').style.display = 'none';
                selectedTextForReview = '';
            }
        });

        document.addEventListener('mousedown', (e) => {
            const floatingBtn = document.getElementById('floatingReviewBtn');
            const content = document.getElementById('articleContent');
            if (floatingBtn && !floatingBtn.contains(e.target) && content && !content.contains(e.target)) {
                floatingBtn.style.display = 'none';
            }
        });

        window.openReviewForSelection = function() {
            if (!loadedArticle) return;
            document.getElementById('floatingReviewBtn').style.display = 'none';
            
            document.getElementById('reviewStatus').value = '3'; // Default to Request Revision
            document.getElementById('feedbackGroup').style.display = 'block';
            
            // Format the selected text as a blockquote
            const quote = `> "${selectedTextForReview}"\n\n`;
            document.getElementById('reviewFeedback').value = quote;
            
            openModal('reviewModal');
            
            // Focus textarea and place cursor at the end
            setTimeout(() => {
                const textarea = document.getElementById('reviewFeedback');
                textarea.focus();
                textarea.setSelectionRange(textarea.value.length, textarea.value.length);
            }, 100);
        };

        await loadArticle();
    </script>
