<script>
    const modal = new bootstrap.Modal(document.getElementById('webhookModal'));

    async function loadWebhooks() {
        try {
            const subs = await apiFetch('/api/webhook');
            const tbody = document.getElementById('webhooksBody');
            tbody.innerHTML = subs.map(s => `
                <tr>
                    <td>${s.id}</td>
                    <td>${s.url}</td>
                    <td>${s.eventsCsv}</td>
                    <td>${s.isActive ? 'Yes' : 'No'}</td>
                    <td>
                        <button class="btn btn-sm btn-warning" onclick="editWebhook(${s.id})">Edit</button>
                        <button class="btn btn-sm btn-danger" onclick="deleteWebhook(${s.id})">Delete</button>
                    </td>
                </tr>
            `).join('');
        } catch (e) {
            console.error('Failed to load webhooks', e);
        }
    }

    function showCreateModal() {
        document.getElementById('webhookForm').reset();
        document.getElementById('webhookId').value = '';
        document.getElementById('webhookModalTitle').textContent = 'Create Webhook';
        modal.show();
    }

    async function editWebhook(id) {
        try {
            const subs = await apiFetch('/api/webhook');
            const sub = subs.find(s => s.id === id);
            if (sub) {
                document.getElementById('webhookId').value = sub.id;
                document.getElementById('url').value = sub.url;
                document.getElementById('eventsCsv').value = sub.eventsCsv;
                document.getElementById('secret').value = sub.secret;
                document.getElementById('isActive').checked = sub.isActive;
                document.getElementById('webhookModalTitle').textContent = 'Edit Webhook';
                modal.show();
            }
        } catch (e) {
            console.error(e);
        }
    }

    async function saveWebhook() {
        const id = document.getElementById('webhookId').value;
        const payload = {
            url: document.getElementById('url').value,
            eventsCsv: document.getElementById('eventsCsv').value,
            secret: document.getElementById('secret').value,
            isActive: document.getElementById('isActive').checked
        };

        try {
            if (id) {
                await apiFetch(`/api/webhook/${id}`, { method: 'PUT', body: JSON.stringify(payload) });
            } else {
                await apiFetch('/api/webhook', { method: 'POST', body: JSON.stringify(payload) });
            }
            modal.hide();
            loadWebhooks();
        } catch (e) {
            alert('Failed to save webhook: ' + e.message);
        }
    }

    async function deleteWebhook(id) {
        if (confirm('Are you sure you want to delete this webhook?')) {
            try {
                await apiFetch(`/api/webhook/${id}`, { method: 'DELETE' });
                loadWebhooks();
            } catch (e) {
                alert('Failed to delete webhook: ' + e.message);
            }
        }
    }

    document.addEventListener('DOMContentLoaded', loadWebhooks);
</script>
