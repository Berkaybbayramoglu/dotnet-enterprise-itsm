    <script type="module">
        import './js/api.js';
        import { bindShellActions, showToast, openModal, closeModal } from './js/ui.js';

        if (!window.api.token) window.location.href = '/login.html';
        bindShellActions();

        let ruleToDelete = null;
        let ruleToEdit = null;
        let lookupData = {};
        let rulesList = [];

        window.openModal = openModal;
        window.closeModal = closeModal;

        async function loadRules() {
            try {
                const rules = await window.api.getAssignmentRules();
                rulesList = rules;
                const tbody = document.querySelector('#rulesTable tbody');
                tbody.innerHTML = '';
                
                if (rules.length === 0) {
                    tbody.innerHTML = `<tr><td colspan="9" style="text-align: center; padding: var(--spacing-xl); color: var(--text-muted);">No assignment rules found.</td></tr>`;
                    return;
                }

                rules.forEach(r => {
                    const tr = document.createElement('tr');
                    
                    const statusBadge = r.isActive 
                        ? `<span class="badge badge-success btn-toggle" style="cursor: pointer;" data-id="${r.id}" title="Click to disable">Active</span>`
                        : `<span class="badge badge-default btn-toggle" style="cursor: pointer;" data-id="${r.id}" title="Click to enable">Inactive</span>`;
                    
                    const projName = lookupData.projects?.find(x => x.id === r.projectId)?.name || 'Herhangi';
                    const catName = lookupData.categories?.find(x => x.id === r.categoryId)?.name || 'Herhangi';
                    const typeName = lookupData.ticketTypes?.find(x => x.id === r.ticketTypeId)?.name || 'Herhangi';
                    const groupName = lookupData.groups?.find(x => x.id === r.targetGroupId)?.name || '—';
                    const userName = lookupData.users?.find(x => x.id === r.targetUserId)?.username || '—';

                    tr.innerHTML = `
                        <td style="font-family: monospace;">${r.sortOrder}</td>
                        <td style="font-weight: 500;">${escapeHtml(r.name)}</td>
                        <td class="text-muted">${escapeHtml(projName)}</td>
                        <td class="text-muted">${escapeHtml(catName)}</td>
                        <td class="text-muted">${escapeHtml(typeName)}</td>
                        <td class="text-muted">${escapeHtml(groupName)}</td>
                        <td class="text-muted">${escapeHtml(userName)}</td>
                        <td>${statusBadge}</td>
                        <td>
                            <button type="button" class="btn btn-ghost btn-edit" style="color: var(--primary); padding: 4px 8px;" data-id="${r.id}">
                                <svg viewBox="0 0 24 24" width="18" height="18" style="fill: currentColor;"><path d="M3 17.25V21h3.75L17.81 9.94l-3.75-3.75L3 17.25zM20.71 7.04c.39-.39.39-1.02 0-1.41l-2.34-2.34c-.39-.39-1.02-.39-1.41 0l-1.83 1.83 3.75 3.75 1.83-1.83z"/></svg>
                            </button>
                            <button type="button" class="btn btn-ghost btn-delete" style="color: var(--danger); padding: 4px 8px;" data-id="${r.id}">
                                <svg viewBox="0 0 24 24" width="18" height="18" style="fill: currentColor;"><path d="M6 19c0 1.1.9 2 2 2h8c1.1 0 2-.9 2-2V7H6v12zM19 4h-3.5l-1-1h-5l-1 1H5v2h14V4z"/></svg>
                            </button>
                        </td>
                    `;
                    tbody.appendChild(tr);
                });
            } catch (err) {
                let errEl = document.querySelector('tbody') || document.querySelector('#overviewCards') || document.querySelector('#board') || document.querySelector('#articleList') || document.querySelector('.content-area');
                if (errEl) {
                    if (errEl.tagName === 'TBODY') {
                        errEl.innerHTML = '<tr><td colspan="100" style="text-align: center; padding: var(--spacing-xl); color: var(--danger);">Veri yüklenemedi — API\'yi kontrol et</td></tr>';
                    } else {
                        errEl.innerHTML = '<div style="padding: 20px; text-align: center; color: var(--danger);">Veri yüklenemedi — API\'yi kontrol et</div>';
                    }
                }

                console.error(err);
                showToast('Error loading rules', 'error');
            }
        }

        document.querySelector('#rulesTable tbody').addEventListener('click', async (e) => {
            const deleteBtn = e.target.closest('.btn-delete');
            const editBtn = e.target.closest('.btn-edit');
            const toggleBtn = e.target.closest('.btn-toggle');
            
            if (deleteBtn) {
                ruleToDelete = Number.parseInt(deleteBtn.dataset.id);
                openModal('confirmModal');
            } else if (editBtn) {
                ruleToEdit = Number.parseInt(editBtn.dataset.id);
                const r = rulesList.find(x => x.id === ruleToEdit);
                if (r) {
                    document.getElementById('rName').value = r.name;
                    document.getElementById('rSort').value = r.sortOrder;
                    document.getElementById('rProject').value = r.projectId || '';
                    document.getElementById('rCategory').value = r.categoryId || '';
                    document.getElementById('rType').value = r.ticketTypeId || '';
                    document.getElementById('rGroup').value = r.targetGroupId || '';
                    document.getElementById('rUser').value = r.targetUserId || '';
                    document.getElementById('rActive').checked = r.isActive;
                    document.getElementById('ruleModalTitle').textContent = 'Edit Assignment Rule';
                    document.getElementById('btnSubmitRule').textContent = 'Save Changes';
                    openModal('createModal');
                }
            } else if (toggleBtn) {
                const id = Number.parseInt(toggleBtn.dataset.id);
                toggleBtn.style.opacity = '0.5';
                try {
                    await window.api.toggleAssignmentRule(id);
                    showToast('Rule status updated');
                    loadRules();
                } catch (err) {
                    console.error(err);
                    showToast('Failed to toggle rule', 'error');
                    toggleBtn.style.opacity = '1';
                }
            }
        });

        document.getElementById('btnConfirmDelete').addEventListener('click', async () => {
            if (!ruleToDelete) return;
            try {
                await window.api.deleteAssignmentRule(ruleToDelete);
                closeModal('confirmModal');
                showToast('Rule deleted successfully');
        loadRules(); // removed await for resilience
            } catch (e) {
                console.error(e);
                showToast('Failed to delete rule', 'error');
            } finally {
                ruleToDelete = null;
            }
        });

        document.getElementById('btnCreateRuleModal').addEventListener('click', () => {
            ruleToEdit = null;
            document.getElementById('createForm').reset();
            document.getElementById('ruleModalTitle').textContent = 'Create Assignment Rule';
            document.getElementById('btnSubmitRule').textContent = 'Create Rule';
            openModal('createModal');
        });

        document.getElementById('createForm').addEventListener('submit', async (e) => {
            e.preventDefault();
            const payload = {
                name: document.getElementById('rName').value.trim(),
                sortOrder: Number.parseInt(document.getElementById('rSort').value) || 10,
                isActive: document.getElementById('rActive').checked,
                projectId: document.getElementById('rProject').value ? Number.parseInt(document.getElementById('rProject').value) : null,
                categoryId: document.getElementById('rCategory').value ? Number.parseInt(document.getElementById('rCategory').value) : null,
                ticketTypeId: document.getElementById('rType').value ? Number.parseInt(document.getElementById('rType').value) : null,
                targetGroupId: document.getElementById('rGroup').value ? Number.parseInt(document.getElementById('rGroup').value) : null,
                targetUserId: document.getElementById('rUser').value ? Number.parseInt(document.getElementById('rUser').value) : null
            };
            
            try {
                if (ruleToEdit) {
                    await window.api.updateAssignmentRule(ruleToEdit, payload);
                    showToast('Rule updated successfully');
                } else {
                    await window.api.createAssignmentRule(payload);
                    showToast('Rule created successfully');
                }
                closeModal('createModal');
                loadRules();
            } catch (e) {
                console.error(e);
                showToast(ruleToEdit ? 'Failed to update rule' : 'Failed to create rule', 'error');
            }
        });

        function escapeHtml(unsafe) {
            return (unsafe || '').toString()
                .replaceAll('&', "&amp;")
                .replaceAll('<', "&lt;")
                .replaceAll('>', "&gt;")
                .replaceAll('"', "&quot;")
                .replaceAll("'", "&#039;");
        }

        // Close modal when clicking outside
        window.onclick = function(event) {
            if (event.target.classList.contains('modal')) {
                event.target.style.display = "none";
            }
        }
        async function initRules() {
            try {
                lookupData = await window.api.request('/lookup');
                const usersList = await window.api.getUsers();
                const groupsList = await window.api.getGroups();
                lookupData.users = usersList;
                lookupData.groups = groupsList;

                const populate = (id, arr) => {
                    const el = document.getElementById(id);
                    arr.forEach(x => {
                        const opt = document.createElement('option');
                        opt.value = x.id; opt.textContent = x.name || x.username || x.id;
                        el.appendChild(opt);
                    });
                };
                populate('rProject', lookupData.projects || []);
                populate('rCategory', lookupData.categories || []);
                populate('rType', lookupData.ticketTypes || []);
                populate('rGroup', lookupData.groups || []);
                populate('rUser', lookupData.users || []);
            } catch(e) {
                console.error("Failed to load lookups", e);
            }
            loadRules();
        }

        document.addEventListener('DOMContentLoaded', initRules);
    </script>
