        <div class="survey-header">
            <h2>Customer Satisfaction</h2>
            <p>How would you rate the resolution of your ticket?</p>
            <div id="ticketInfo" class="ticket-badge" style="display:none;"></div>
        </div>

        <form id="surveyForm">
            <div class="rating-group">
                <div class="rating">
                    <input type="radio" id="star5" name="rating" value="5" required />

                <textarea id="comment" class="form-control" placeholder="Tell us more about your experience..."></textarea>
            </div>
            
            <button type="submit" class="btn btn-primary" style="width: 100%; padding: 12px; font-size: 16px;">Submit Feedback</button>

    <script type="module">
        import { injectShell } from './js/layout.js?v=10';
        import './js/api.js';
        import { showToast } from './js/ui.js';
        injectShell();
        
        // Survey might be public or require token depending on implementation
        // For now we keep the existing logic where it redirects if no token, 
        // though typically CSATs use one-time tokens in the URL.
        if (!window.api.token) {
            // Optional: comment out redirect if survey is public
            // window.location.href = '/login.html'; 
        }

        const urlParams = new URLSearchParams(window.location.search);
        const ticketId = urlParams.get('id');
        const ticketInfo = document.getElementById('ticketInfo');

        if (!ticketId) {
            document.getElementById('surveyContainer').innerHTML = `
                <div class="success-state" style="color: var(--danger);">
                    <svg viewBox="0 0 24 24" width="48" height="48" style="fill: currentColor; margin-bottom: 16px;"><path d="M12 2C6.48 2 2 6.48 2 12s4.48 10 10 10 10-4.48 10-10S17.52 2 12 2zm1 15h-2v-2h2v2zm0-4h-2V7h2v6z"/></svg>
                    <h3 style="margin:0; font-size: 20px;">Invalid Link</h3>
                    <p style="margin:0; color: var(--text-muted);">No ticket ID provided.</p>
                </div>
            `;
        } else {
            ticketInfo.textContent = `Ticket #${ticketId}`;
            ticketInfo.style.display = 'inline-block';
        }

        document.getElementById('surveyForm')?.addEventListener('submit', async (e) => {
            e.preventDefault();
            
            const ratingInput = document.querySelector('input[name="rating"]:checked');
            if (!ratingInput) {
                showToast("Please select a rating.", "error");
                return;
            }

            const rating = Number.parseInt(ratingInput.value);
            const comment = document.getElementById('comment').value;
            
            const btn = e.target.querySelector('button[type="submit"]');
            const originalText = btn.textContent;
            btn.textContent = 'Submitting...';
            btn.disabled = true;

            try {
                await window.api.request(`/Ticket/${ticketId}/survey`, {
                    method: 'POST',
                    body: JSON.stringify({ rating, comment })
                });
                
                document.getElementById('surveyContainer').innerHTML = `
                    <div class="success-state">
                        <div class="success-icon">
                            <svg viewBox="0 0 24 24" width="32" height="32" style="fill: currentColor;"><path d="M9 16.17L4.83 12l-1.42 1.41L9 19 21 7l-1.41-1.41z"/></svg>
                        </div>
                        <h3 style="margin:0; font-size: 24px; color: var(--text-main);">Thank You!</h3>
                        <p style="margin:0; color: var(--text-muted); text-align: center; max-width: 300px;">Your feedback has been recorded. We appreciate you taking the time to help us improve.</p>
                        <a href="/dashboard.html" class="btn btn-primary" style="margin-top: var(--spacing-lg);">Return to Dashboard</a>
                    </div>
                `;
            } catch (err) {
                console.error(err);
                showToast('Failed to submit feedback', 'error');
                btn.textContent = originalText;
                btn.disabled = false;
            }
        });
    </script>
